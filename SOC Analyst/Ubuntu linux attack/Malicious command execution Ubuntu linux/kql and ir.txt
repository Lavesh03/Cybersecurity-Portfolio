Kql:-
1.let window = 10m;
let RemoteLogins = AttackLogs_CL
    | where RawData has "Accepted password" and RawData has "ssh2"
    | extend SrcIP = extract(@"from (\S+)", 1, RawData)
    | project LoginTime = TimeGenerated, Host, SrcIP;
let SudoCmds = AttackLogs_CL
    | where RawData has "COMMAND="
    | extend Cmd = extract(@"COMMAND=(.*)", 1, RawData),
             User = extract(@"sudo:\s*(\S+)\s*:", 1, RawData)
    | project CmdTime = TimeGenerated, Host, User, Cmd;
RemoteLogins
| join kind=inner (SudoCmds) on Host
| where CmdTime between (LoginTime .. (LoginTime + window))
| summarize LoginTime = min(LoginTime) by CmdTime, Host, SrcIP, User, Cmd
| order by CmdTime asc

2.AttackLogs_CL
| where RawData contains "COMMAND="
| extend Cmd = extract(@"COMMAND=(.*)", 1, RawData)
| extend Tactic = case(
    Cmd has "/etc/shadow", "Credential Access",
    Cmd has "find" and Cmd has "/bin/sh", "Privilege Escalation",
    Cmd has "ss -tulpn" or Cmd has "ps aux", "Discovery",
    Cmd has "systemctl enable" or Cmd has "sandbox-check", "Persistence",
    "Other")
| where Tactic != "Other"
| summarize Count = count() by Tactic

3.AttackLogs_CL
| where RawData contains "sandbox-audit.service"
| project TimeGenerated, Host, RawData
| order by TimeGenerated desc
